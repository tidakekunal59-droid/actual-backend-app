+const mongoose = require('mongoose');
+
+// Simple Lock Schema to handle concurrency
+const lockSchema = new mongoose.Schema({
+  resource: { type: String, required: true, unique: true }, // e.g., 'vehicle:123'
+  holder: { type: String, required: true },
+  createdAt: { type: Date, default: Date.now, expires: 60 } // Auto-expire locks after 60s
+});
+
+const Lock = mongoose.model('Lock', lockSchema);
+
+module.exports = {
+  acquireLock: async (resource, holder) => {
+    try {
+      await Lock.create({ resource, holder });
+      return true;
+    } catch (err) {
+      if (err.code === 11000) return false; // Duplicate key error = locked
+      throw err;
+    }
+  },
+
+  releaseLock: async (resource, holder) => {
+    await Lock.deleteOne({ resource, holder });
+  }
+};
